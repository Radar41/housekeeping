    ollivier_ricci_curvature double precision,
    CONSTRAINT atlas_similarity_edge_cosine_range CHECK (((cosine >= ('-1.0'::numeric)::double precision) AND (cosine <= (1.0)::double precision))),
    CONSTRAINT atlas_similarity_edge_distinct_docs CHECK ((doc_id_a <> doc_id_b)),
    CONSTRAINT atlas_similarity_edge_rank_positive CHECK ((edge_rank > 0))
);


--
-- Name: atlas_similarity_edge_canonical; Type: VIEW; Schema: atlas; Owner: -
--

CREATE VIEW atlas.atlas_similarity_edge_canonical AS
 SELECT e.edge_id,
    e.edge_scope,
    e.root_id,
    e.doc_id_a,
    e.doc_id_b,
    e.chunk_id_a,
    e.chunk_id_b,
    e.model,
    e.cosine,
    e.edge_rank,
    e.shard_index,
    e.shard_count,
    e.metadata,
    e.created_at,
    e.updated_at,
    e.angular_distance,
    e.ollivier_ricci_curvature
   FROM ((atlas.atlas_similarity_edge e
     JOIN atlas.atlas_document da ON ((da.document_id = e.doc_id_a)))
     JOIN atlas.atlas_document db ON ((db.document_id = e.doc_id_b)))
  WHERE (((da.version_status IS NULL) OR (da.version_status = 'canonical'::text)) AND ((db.version_status IS NULL) OR (db.version_status = 'canonical'::text)));

-- Name: VIEW atlas_similarity_edge_canonical; Type: COMMENT; Schema: atlas; Owner: -

COMMENT ON VIEW atlas.atlas_similarity_edge_canonical IS 'Gate 3 (signed 2026-07-13): canonical-only curvature scope. Pass 2 communities inherit this scope explicitly; non-canonical edges are geometrically degenerate under the version-status collapse and carry curvature_out_of_scope in metadata if a legacy run ever computed them.';
